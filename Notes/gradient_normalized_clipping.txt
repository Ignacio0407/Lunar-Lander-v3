# Original gradients:

# Paso 1: Calculate total norm
# norm = sqrt(5.2^2 + 150.3^2 + 200.1^2 + ... + 50.2^2)
# norm = sqrt(27.04 + 22590.09 + 40040.01 + ... + 2520.04)
# norm ≈ 250.7  ← Really high

# Step 2: ¿norm > max_norm? (250.7 > 10)
# Yes → Calculate scaling factor
scale_factor = max_norm / norm = 10 / 250.7 ≈ 0.0399

# Step 3: Scale all gradients
layer1_grad_clipped = layer1_grad * 0.0399
# = [-0.207, 5.997, -7.984, 1.217]  ← Direction preserved!
